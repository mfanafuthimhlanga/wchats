  def test_deployment_checklist_completes():
    BASE_URL = os.environ.get("BASE_URL", "http://localhost:8000")
    API_KEY = os.environ.get("API_KEY", "")
    AGENT_ID = os.environ.get("E2E_AGENT_ID", "")  # pre-provisioned agent
    headers = {"X-API-Key": API_KEY}

    # Trigger checklist run
    trigger_resp = requests.post(f"{BASE_URL}/api/v1/agents/{AGENT_ID}/checklist-runs", headers=headers, timeout=30)
    assert trigger_resp.status_code == 202
    run_id_or_task_id = trigger_resp.json().get("checklist_run_id")
    assert run_id_or_task_id

    # Poll for completion (300s deadline, 3s interval)
    deadline = time.time() + 300
    complete_run = None
    while time.time() < deadline:
        poll_resp = requests.get(f"{BASE_URL}/api/v1/agents/{AGENT_ID}/checklist-runs", headers=headers, timeout=30)
        assert poll_resp.status_code == 200
        runs = poll_resp.json().get("runs", [])
        for run in runs:
            if run.get("status") in ("complete", "failed"):
                complete_run = run
                break
        if complete_run:
            break
        time.sleep(3)

    assert complete_run is not None, "Checklist run did not complete within 300s"
    assert complete_run["status"] == "complete", f"Run status: {complete_run['status']}"
    assert complete_run["recommendation"] in ["ship", "ship_with_warnings", "block"]
    assert isinstance(complete_run["warnings"], list)
    assert isinstance(complete_run["all_warnings_acknowledged"], bool)

    # If approvable, test approval flow
    if complete_run["recommendation"] != "block":
        # Acknowledge any warnings
        if complete_run["recommendation"] == "ship_with_warnings":
            warning_ids = [w["warning_id"] for w in complete_run["warnings"]]
            ack_resp = requests.post(
                f"{BASE_URL}/api/v1/agents/{AGENT_ID}/checklist-runs/{complete_run['id']}/acknowledge",
                json={"warning_ids": warning_ids},
                headers=headers,
                timeout=30,
            )
            assert ack_resp.status_code == 200

        # Approve
        approve_resp = requests.post(
            f"{BASE_URL}/api/v1/agents/{AGENT_ID}/approve-deployment",
            json={"checklist_run_id": complete_run["id"]},
            headers=headers,
            timeout=30,
        )
        assert approve_resp.status_code == 200
        data = approve_resp.json()
        assert data["deployed"] is True
        assert "iframe_snippet" in data
        assert "widget.veridian.app" in data["iframe_snippet"]
  </action>
