<!-- From 03-CONTEXT.md locked RRF SQL (verbatim) -->
WITH vector_ranked AS (
    SELECT e.chunk_id, c.content, c.document_id,
           1 - (e.vector <=> %(query_vector)s::vector) AS cosine_score,
           ROW_NUMBER() OVER (ORDER BY e.vector <=> %(query_vector)s::vector) AS rank
    FROM embeddings e JOIN chunks c ON c.id = e.chunk_id
    ORDER BY e.vector <=> %(query_vector)s::vector
    LIMIT %(vector_k)s
),
bm25_ranked AS (
    SELECT c.id AS chunk_id, c.content, c.document_id,
           ts_rank_cd(to_tsvector('english', c.content), plainto_tsquery('english', %(query)s)) AS bm25_score,
           ROW_NUMBER() OVER (ORDER BY ts_rank_cd(...) DESC) AS rank
    FROM chunks c
    WHERE to_tsvector('english', c.content) @@ plainto_tsquery('english', %(query)s)
    LIMIT %(bm25_k)s
),
fused AS (
    SELECT COALESCE(v.chunk_id, b.chunk_id) AS chunk_id,
           COALESCE(v.content, b.content) AS content,
           COALESCE(v.document_id, b.document_id) AS document_id,
           COALESCE(1.0 / (60.0 + v.rank), 0.0) + COALESCE(1.0 / (60.0 + b.rank), 0.0) AS rrf_score,
           v.cosine_score, b.bm25_score, v.rank AS vector_rank, b.rank AS bm25_rank
    FROM vector_ranked v FULL OUTER JOIN bm25_ranked b ON v.chunk_id = b.chunk_id
)
SELECT chunk_id, content, document_id, rrf_score,
       cosine_score, bm25_score, vector_rank, bm25_rank
FROM fused ORDER BY rrf_score DESC LIMIT %(final_k)s

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| query_text → psycopg2 | User query string passed as parameter; must use parameterized SQL only |
| query_vector → psycopg2 | Embedding vector stringified as str(list) — no SQL injection possible (numeric list) |
| Voyage API → service | External API call; response shapes validated by namedtuple access |
| Cohere API → service | External API call; only called as fallback |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-04 | Injection | bm25_search — plainto_tsquery | mitigate | plainto_tsquery normalizes input — does not execute arbitrary SQL; no string interpolation |
| T-03-05 | Injection | vector_search — query_vector | accept | query_vector is str(list[float]) — purely numeric; no SQL injection risk |
| T-03-06 | Information Disclosure | rerank — log statements | mitigate | Only log error_type (exception class name), not query text or vector values |
| T-03-07 | Information Disclosure | COHERE_API_KEY in _cohere_rerank | mitigate | Key read from settings only; never passed to logs; fallback imports cohere lazily |
| T-03-08 | Tampering | RetrievalStrategy JSONB injection | mitigate | model_validate with ConfigDict(extra="ignore") — unknown keys silently dropped; numeric fields type-validated |
</threat_model>
