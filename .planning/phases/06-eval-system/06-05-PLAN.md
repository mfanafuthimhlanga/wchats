## Objective

Task: run_eval_suite_beat
  @celery_app.task(bind=True, acks_late=True, max_retries=1, default_retry_delay=60, queue="runtime", name="app.worker.tasks.runtime.eval.run_eval_suite_beat")
  def run_eval_suite_beat(self) -> dict:
    Docstring: "Beat-triggered dispatcher: find all status='ready' agents and dispatch run_eval_suite per agent."
    
    with get_sync_db() as db:
        agents = db.execute(select(Agent).where(Agent.status == "ready")).scalars().all()
    
    dispatched = 0
    for agent in agents:
        run_eval_suite.apply_async(kwargs={"agent_id": str(agent.id)}, queue="runtime")
        dispatched += 1
    
    log.info("run_eval_suite_beat.dispatched", count=dispatched)
    return {"dispatched": dispatched}
</action>

Task: run_eval_suite
  @celery_app.task(bind=True, acks_late=True, max_retries=2, default_retry_delay=30, queue="runtime", name="app.worker.tasks.runtime.eval.run_eval_suite")
  def run_eval_suite(self, agent_id: str) -> dict:
    Docstring: "Per-agent eval run. Creates Neon branch, runs Ragas 0.4.x eval, promotes verified_qa, deletes branch in finally (D-10). Receives agent_id str — no conn_str in args (CTL-08 / D-18)."
    
    Idempotency guard: check eval_runs on tenant DB for a 'running' eval run created within the last 10 minutes for this agent (uses eval_runs.kind = f"m6:{agent_id}" as the idempotency key). If found, return {"status": "already_running"}.
    
    1. Fetch agent from control DB: with get_sync_db() as db: agent = db.get(Agent, agent_id)
       if agent is None or not agent.neon_project_id or not agent.neon_connection_string: log.error + return {}
    
    2. Decrypt conn_str: conn_str = fernet_decrypt(agent.neon_connection_string)
    
    3. Fetch eval scenarios from tenant DB (psycopg2): SELECT id, source, question, reference_answer, retrieved_contexts FROM eval_scenarios WHERE reference_answer != '' ORDER BY RANDOM() LIMIT 30
       Also mine new scenarios and store them: mine_production_scenarios(agent_id, conn_str, control_db) + store_scenarios(mined, conn_str)
    
    4. If no scenarios: log.warning("run_eval_suite.no_scenarios") + return {"status": "no_scenarios"}
    
    5. Insert eval_run row: run_id = str(uuid.uuid4()); INSERT INTO eval_runs (id, kind, started_at, status) VALUES (%s, %s, NOW(), 'running') using psycopg2 on conn_str (production branch — record goes to prod DB)
    
    6. Create Neon branch: branch_id, branch_conn_str = create_branch(agent.neon_project_id, f"eval-{run_id}")
       wait_for_neon_ready(branch_conn_str)
    
    7. try/finally block with delete_branch in finally:
       branch_id_for_finally = branch_id  # captured before try
       try:
           - Build scenarios list for Ragas: filter out rows with empty reference_answer; attach agent_response="" (will be filled by running the agent — for M6 use reference_answer as proxy agent_response to test the eval harness)
           - results = run_ragas_eval(scenarios, branch_conn_str)
           - write_eval_results(run_id, results["scores"], conn_str)  # write to prod branch
           - promoted = promote_to_verified_qa(scenarios, results["scores"], conn_str)
           - update_eval_run_status(run_id, "complete", finished_at=True, conn_str=conn_str)
           - log.info("run_eval_suite.complete", agent_id=agent_id, run_id=run_id, promoted=promoted)
           - return {"run_id": run_id, "scenario_count": len(scenarios), "promoted": promoted}
       except Exception as exc:
           - update_eval_run_status(run_id, "failed", finished_at=True, conn_str=conn_str)
           - log + retry or return {}
       finally:
           - try: delete_branch(agent.neon_project_id, branch_id_for_finally)
           - except Exception as del_exc: log.warning("run_eval_suite.branch_delete_failed", error=str(del_exc))
</action>
