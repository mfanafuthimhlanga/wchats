Add two rate-limiting controls to widget.py:

F2 (CRITICAL): The public GET /widget/{agent_id}/config endpoint mints a fresh JWT
on every call with no per-IP throttle — enabling unlimited JWT harvest. Add
_check_config_rate_limit helper keyed on client IP (rate:config:{ip}:{bucket}) with

redis_client=Depends(get_async_redis) parameters.

connection exhaustion DoS. Add per-agent_id slot management via Redis SETNX
(_acquire_sse_slot / _release_sse_slot) capped at _MAX_CONCURRENT_SSE_PER_AGENT=50,
plus asyncio.timeout(120) hard cap on the SSE generator loop.

Purpose: F2 removes a trivial attack path for JWT harvest and cost-bomb attacks.
F8 prevents connection exhaustion from the same public-facing surface. Both fix
widget.py only with no downstream dependencies.

Output: Modified widget.py with config rate limiter and SSE slot cap;
extended test_widget_routes.py with 5 new security tests.

  _MAX_CONCURRENT_SSE_PER_AGENT = 50

  async def _acquire_sse_slot(agent_id: str, job_id: str, redis: Redis) -> bool:
      count_key = f"sse:count:{agent_id}"
      slot_key  = f"sse:slot:{agent_id}:{job_id}"
      acquired = await redis.set(slot_key, "1", nx=True, ex=150)
      if not acquired:
          return False
      count = await redis.incr(count_key)
      if count == 1:
          await redis.expire(count_key, 3600)
      if count > _MAX_CONCURRENT_SSE_PER_AGENT:
          await redis.delete(slot_key)
          await redis.decr(count_key)
          return False
      return True

  async def _release_sse_slot(agent_id: str, job_id: str, redis: Redis) -> None:
      slot_key = f"sse:slot:{agent_id}:{job_id}"
      deleted = await redis.delete(slot_key)
      if deleted:
          await redis.decr(f"sse:count:{agent_id}")
