  def provision_neon(self, tenant_id: str, agent_id: str) -> dict:
  1. with get_sync_db() as db:
  2.   agent = db.get(Agent, agent_id)
  3.   if not agent: raise ValueError(f"Agent {agent_id} not found")

  4.   # Idempotency guard — if already provisioned, skip and return
  5.   if agent.neon_project_id:
  6.     return {"agent_id": agent_id, "project_id": agent.neon_project_id}

  7.   # Mark agent as provisioning; fetch job
  8.   agent.status = "provisioning"
  9.   job = db.query(Job).filter(Job.agent_id == agent_id, Job.status != "complete").first()
  10.  if not job: return {"agent_id": agent_id, "project_id": agent.neon_project_id}
  11.  job.status = "running"; job.started_at = datetime.now(timezone.utc); db.commit()

  12.  # Emit job.started as the first event of this chain (before any Neon work)
  13.  emit(job.id, "job.started", {"agent_id": agent_id}, db, _redis)

  14.  emit(job.id, "neon.project.creating", {"agent_id": agent_id}, db, _redis)

  15.  try:
  16.    result = create_neon_project(agent_id)  # {"id": project_id, "pooled_uri": ..., "direct_uri": ...}
  17.  except Exception as exc:
         # Determine if 4xx (fatal) or 5xx/timeout (retry)
         status_code = getattr(exc, "status_code", None) or getattr(getattr(exc, "response", None), "status_code", None)
         if status_code and 400 <= status_code < 500:
           # Fatal: 4xx
           agent.status = "failed"; job.status = "failed"; job.error = str(exc); job.finished_at = datetime.now(timezone.utc); db.commit()
           emit(job.id, "job.failed", {"error": str(exc)}, db, _redis)
           return
         # Retriable: 5xx or timeout
         raise self.retry(exc=exc, countdown=2 ** self.request.retries)

  18.  project_id = result["id"]

  19.  # CRITICAL: Write project_id IMMEDIATELY before any further work (idempotency safety)
  20.  agent.neon_project_id = project_id; db.commit()

  21.  # Encrypt both pooled and direct URIs; store separately
  22.  pooled_encrypted = fernet_encrypt(result["pooled_uri"])
  23.  direct_encrypted = fernet_encrypt(result["direct_uri"])
  24.  agent.neon_connection_string = pooled_encrypted        # pooled — for application traffic
  25.  agent.neon_direct_connection_string = direct_encrypted  # direct — for Alembic migrations only
  26.  db.commit()

  27.  emit(job.id, "neon.project.ready", {"project_id": project_id}, db, _redis)

  28.  return {"agent_id": agent_id, "project_id": project_id}

  def apply_migrations(self, result: dict) -> None:
  1. agent_id = result["agent_id"]
  2. with get_sync_db() as db:
  3.   agent = db.get(Agent, agent_id)

  4.   # Idempotency guard — if agent is already ready, this task has already completed
  5.   if agent.status == "ready": return

  6.   job = db.query(Job).filter(Job.agent_id == agent_id, Job.status == "running").first()
  7.   if not job: return  # job was already completed on a prior attempt

  8.   # Fetch and decrypt DIRECT connection string from control DB (NOT from task args)
  9.   direct_conn_string = fernet_decrypt(agent.neon_direct_connection_string)

  10.  emit(job.id, "migrations.running", {"agent_id": agent_id}, db, _redis)

  11.  # Connection probe — wait for Neon compute to be query-ready
  12.  try:
  13.    wait_for_neon_ready(direct_conn_string)
  14.  except RuntimeError as exc:
  15.    # Probe exhausted — retry task
  16.    raise self.retry(exc=exc, countdown=2 ** self.request.retries)

  17.  # Run migrations
  18.  try:
  19.    run_tenant_migrations(direct_conn_string)
  20.  except Exception as exc:
  21.    # Migration errors are fatal — log for manual inspection, do not retry
  22.    agent.status = "failed"
  23.    job.status = "failed"; job.error = str(exc); job.finished_at = datetime.now(timezone.utc)
  24.    db.commit()
  25.    emit(job.id, "job.failed", {"error": str(exc)}, db, _redis)
  26.    return  # Do not raise; chain ends here

  27.  # Record schema version and mark agent ready
  28.  revision = get_current_alembic_revision(direct_conn_string)
  29.  agent.schema_version = revision
  30.  agent.status = "ready"
  31.  job.status = "complete"; job.finished_at = datetime.now(timezone.utc)
  32.  db.commit()

  33.  emit(job.id, "migrations.complete", {"schema_version": revision}, db, _redis)
  34.  emit(job.id, "job.complete", {"agent_id": agent_id, "schema_version": revision}, db, _redis)
