    ARCHITECTURAL EXCEPTION NOTE (must appear in module docstring):
    The psycopg2 INSERT in the POST route is an accepted exception to the "FastAPI never does work inline"
    rule (CLAUDE.md). Document row creation is a prerequisite for chain dispatch — the chain requires
    document IDs to exist in the tenant DB before it can process them. This insertion cannot be
    Celery-delegated without creating a circular dependency (Celery task that creates the row for which
    Celery was invoked). The INSERT is bounded: one row per uploaded file, no ML model calls, no
    external network calls. The psycopg2.connect() call MUST include connect_timeout=5 to bound
    blocking on the route's async event loop thread.

    Route 1 — POST /agents/{agent_id}/documents:

    @router.post("/agents/{agent_id}/documents", status_code=202, response_model=DocumentUploadResponse)
    async def upload_documents(
        agent_id: UUID,
        files: list[UploadFile] = File(default=[]),
        urls: list[str] = Form(default=[]),
        db: AsyncSession = Depends(get_async_db),
        tenant: Tenant = Depends(get_current_tenant),
    ) -> DocumentUploadResponse:

    Body order:
    1. Validate agent exists AND belongs to tenant AND status='ready' AND deleted_at IS NULL:
       result = await db.execute(
           select(Agent).where(
               Agent.id == agent_id,
               Agent.tenant_id == tenant.id,
               Agent.deleted_at.is_(None),
           )
       )
       agent = result.scalar_one_or_none()
       if agent is None: raise HTTPException(404, "Agent not found")
       if agent.status != "ready": raise HTTPException(409, detail=f"Agent is not ready (status={agent.status})")

    2. Validate at least one source:
       if not files and not urls: raise HTTPException(422, "At least one file or URL is required")

    3. Validate file types and sizes (BEFORE writing to disk):
       max_bytes = settings.MAX_UPLOAD_SIZE_MB * 1024 * 1024
       for f in files:
           ext = Path(f.filename or "").suffix.lower()
           if ext not in ALLOWED_EXTENSIONS:
               raise HTTPException(415, f"Unsupported file type: {ext}")
           # Read into memory once to check size; for M2 this is acceptable (files are small)
           contents = await f.read()
           if len(contents) > max_bytes:
               raise HTTPException(413, f"File {f.filename} exceeds MAX_UPLOAD_SIZE_MB={settings.MAX_UPLOAD_SIZE_MB}")
           f._cached_contents = contents  # stash for the write loop below; reset .file later if you re-use

       (Implementation note: UploadFile.read() consumes the stream; cache contents in a local dict keyed by index to avoid awaiting twice. DO NOT call await f.seek(0) without verifying it works on starlette's SpooledTemporaryFile.)

    4. Fetch + decrypt pooled tenant conn (for the document INSERT — FastAPI does this sync via psycopg2 to keep the txn cohesive with the chain dispatch):
       tenant_conn_str = fernet_decrypt(agent.neon_connection_string)

    5. Save files to local temp and INSERT document rows in tenant DB. Use a single psycopg2 connection in a transaction with connect_timeout=5:
       upload_dir = Path(tempfile.gettempdir()) / "vrd-uploads" / str(agent.id)
       upload_dir.mkdir(parents=True, exist_ok=True)
       document_ids: list[str] = []
       tenant_conn = psycopg2.connect(tenant_conn_str, connect_timeout=5)
       try:
           with tenant_conn.cursor() as cur:
               for idx, f in enumerate(files):
                   doc_id = str(uuid.uuid4())
                   ext = Path(f.filename).suffix.lower()
                   local_path = upload_dir / f"{doc_id}{ext}"
                   # Reuse cached contents from validation pass
                   contents = cached_contents[idx]
                   local_path.write_bytes(contents)
                   cur.execute(
                       """
                       INSERT INTO documents (id, source_type, source_uri, title)
                       VALUES (%s, %s, %s, %s)
                       """,
                       (doc_id, ext.lstrip("."), f.filename, f.filename),
                   )
                   document_ids.append(doc_id)
               for url_str in urls:
                   doc_id = str(uuid.uuid4())
                   cur.execute(
                       """
                       INSERT INTO documents (id, source_type, source_uri, title)
                       VALUES (%s, %s, %s, %s)
                       """,
                       (doc_id, "url", url_str, url_str),
                   )
                   document_ids.append(doc_id)
           tenant_conn.commit()
       finally:
           tenant_conn.close()

    6. Create job row in control DB:
       job = Job(tenant_id=tenant.id, agent_id=agent.id, kind="ingest_documents", status="pending")
       db.add(job)
       await db.commit()
       await db.refresh(job)

    7. Dispatch Celery chain (mirror agents.py pattern with ctx headers):
       ctx = get_contextvars()
       chain(
           parse_documents.s(str(tenant.id), str(agent.id), str(job.id), document_ids),
           chunk_documents.s(),
           generate_metadata.s(),
           embed_and_migrate.s(),
       ).apply_async(
           queue="pipeline",
           headers={"request_id": ctx.get("request_id", "")},
       )

    8. Return:
       return DocumentUploadResponse(
           job_id=job.id,
           document_ids=[UUID(d) for d in document_ids],
           status="pending",
           events_url=f"/jobs/{job.id}/events",
       )

|----------|-------------|
| HTTP client → FastAPI route | Untrusted multipart input; mitigated by file-type whitelist + size cap + agent ownership filter |

| Route → Celery chain | Task args constructed by route; mitigated by NEVER passing conn string (only IDs) |
| Local filesystem → worker | Uploaded files saved to local temp; mitigated by per-agent subdirectory + uuid4 filenames |
| FastAPI route → tenant DB (psycopg2 inline) | Accepted architectural exception: document row creation is bounded (no ML, no external calls). connect_timeout=5 limits blocking. |
