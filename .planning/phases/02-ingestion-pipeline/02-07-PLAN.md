Wave 7 turns the M2 chain into a demonstrable artifact: a real PDF, an automated demo script, and an E2E test that proves the system works against actual Anthropic + Voyage APIs.

Three core responsibilities:
1. **The demo PDF.** A real business PDF (≤500KB, ≥1 table) committed to the repo at `apps/api/tests/fixtures/demo_business.pdf`. This single file proves ING-03 (table-to-Markdown preservation) when its content shows up correctly in the `chunks` table. Procurement is a human task — no synthetic PDFs.
2. **The demo script.** `scripts/demo_m2.sh` mirrors `scripts/demo_m1.sh`: it assumes docker-compose is up, an agent is provisioned (it provisions one if not), uploads the PDF, streams the SSE event log to the terminal, then prints SELECT results from the tenant `chunks`, `chunk_metadata`, and `entities` tables. Exits 0 on success.
3. **The E2E test.** `tests/integration/test_ingestion_e2e.py` is gated by `INGESTION_E2E_ENABLED=1` (mirroring M1's nightly E2E gate). When enabled, it uploads the demo PDF against a real tenant DB and real external APIs, asserts on observable outcomes (chunks > 5, embedding dim = 1024, ≥1 entity, table chunk present), and re-runs to prove idempotency end-to-end.

This is also the only plan in M2 that contains a human checkpoint — for sourcing the PDF and visually verifying that the SSE event stream is correct.

    def test_real_pdf_ingestion_end_to_end(test_agent_and_job, db_session):
        """ING-10: upload demo_business.pdf, verify chunks/metadata/embeddings/entities populate."""
        agent_id, job_id, tenant_id = test_agent_and_job

        pdf_path = Path("apps/api/tests/fixtures/demo_business.pdf")
        assert pdf_path.exists(), f"Demo PDF not found at {pdf_path}"
        assert pdf_path.stat().st_size < 500_000, "Demo PDF must be < 500KB"

        # POST the PDF (use httpx synchronous; same Postgres/Redis as integration suite)
        import httpx
        with open(pdf_path, "rb") as f:
            response = httpx.post(
                f"http://localhost:8000/api/v1/agents/{agent_id}/documents",
                headers={"X-API-Key": os.environ["API_KEY"]},  # provisioned by fixture
                files={"files": (pdf_path.name, f, "application/pdf")},
                timeout=30,
            )
        assert response.status_code == 202, response.text
        body = response.json()
        document_id = body["document_ids"][0]

        # Wait for chain completion by polling job status (max 300s)
        # In integration mode, CELERY_TASK_ALWAYS_EAGER may be False (real workers).
        # Reuse a wait helper from existing tests or inline:
        import time
        deadline = time.time() + 300
        while time.time() < deadline:
            cur = db_session.execute("SELECT status FROM jobs WHERE id = %s", (body["job_id"],))
            row = cur.fetchone()
            if row and row[0] in ("complete", "failed"):
                break
            time.sleep(2)
        assert row and row[0] == "complete", f"Job did not complete; final status={row}"

        # Inspect tenant DB
        from app.core.security import fernet_decrypt
        from app.models.agent import Agent
        import psycopg2
        agent = db_session.get(Agent, agent_id)
        conn_str = fernet_decrypt(agent.neon_connection_string)
        with psycopg2.connect(conn_str) as conn, conn.cursor() as cur:
            cur.execute("SELECT COUNT(*) FROM chunks WHERE document_id = %s", (document_id,))
            chunk_count = cur.fetchone()[0]
            assert chunk_count > 5, f"Expected >5 chunks, got {chunk_count}"

            cur.execute("SELECT COUNT(*) FROM chunk_metadata cm JOIN chunks c ON c.id = cm.chunk_id WHERE c.document_id = %s", (document_id,))
            metadata_count = cur.fetchone()[0]
            assert metadata_count == chunk_count, f"chunk_metadata count {metadata_count} != chunks {chunk_count}"

            cur.execute("SELECT COUNT(*) FROM embeddings e JOIN chunks c ON c.id = e.chunk_id WHERE c.document_id = %s", (document_id,))

        """ING-09 + ING-10: dispatching the chain twice for the same document produces identical row counts."""
        agent_id, job_id, tenant_id = test_agent_and_job

            embedding_count_before = cur.fetchone()[0]

        # Step 3: re-dispatch the same chain manually using the four task functions directly
        # (avoids creating a second document row; tests idempotency at the task layer)
        from celery import chain as celery_chain
        from app.worker.tasks.pipeline.parse import parse_documents
        from app.worker.tasks.pipeline.chunk import chunk_documents
        from app.worker.tasks.pipeline.metadata import generate_metadata
        from app.worker.tasks.pipeline.embed import embed_and_migrate

        # Create a new job row for the re-dispatch so event emission has a valid job_id
        from app.models.job import Job
        second_job = Job(tenant_id=tenant_id, agent_id=agent_id, kind="ingest_documents", status="pending")
        db_session.add(second_job)
        db_session.commit()
        db_session.refresh(second_job)
        second_job_id = str(second_job.id)

        celery_chain(
            parse_documents.s(str(tenant_id), str(agent_id), second_job_id, [document_id]),
            chunk_documents.s(),
            generate_metadata.s(),
            embed_and_migrate.s(),
        ).apply_async(queue="pipeline")

        # Wait for second chain to complete (max 300s)
        deadline = time.time() + 300
        while time.time() < deadline:
            cur = db_session.execute("SELECT status FROM jobs WHERE id = %s", (second_job_id,))
            row = cur.fetchone()
            if row and row[0] in ("complete", "failed"):
                break
            time.sleep(2)
        assert row and row[0] == "complete", f"Second run did not complete; status={row}"

            chunk_count_after = cur.fetchone()[0]

        )
