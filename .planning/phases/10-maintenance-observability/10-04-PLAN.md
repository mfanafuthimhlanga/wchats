## Task 1 — AlertsBanner component

Create `apps/admin/app/agents/[id]/components/AlertsBanner.tsx`:

```tsx
"use client";
import { useEffect, useState } from "react";

interface Alert {
  id: string;
  alert_type: string;
  severity: string;
  message: string;
  triggered_at: string;
}

interface Props {
  agentId: string;
  apiKey: string;
}

export function AlertsBanner({ agentId, apiKey }: Props) {
  const [alerts, setAlerts] = useState<Alert[]>([]);

  const fetchAlerts = async () => {
    try {
      const res = await fetch(`/api/v1/agents/${agentId}/alerts`, {
        headers: { "X-API-Key": apiKey },
      });
      if (res.ok) setAlerts(await res.json());
    } catch {}
  };

  useEffect(() => {
    fetchAlerts();
    const id = setInterval(fetchAlerts, 30_000);
    return () => clearInterval(id);
  }, [agentId]);

  if (alerts.length === 0) return null;

  const resolve = async (alertId: string) => {
    await fetch(`/api/v1/agents/${agentId}/alerts/${alertId}/resolve`, {
      method: "POST",
      headers: { "X-API-Key": apiKey },
    });
    setAlerts((prev) => prev.filter((a) => a.id !== alertId));
  };

  return (
    <div className="mb-4 space-y-2">
      {alerts.map((a) => (
        <div
          key={a.id}
          className={`flex items-start justify-between rounded-lg px-4 py-3 text-sm ${
            a.severity === "critical"
              ? "bg-red-50 border border-red-200"
              : "bg-amber-50 border border-amber-200"
          }`}
        >
          <div>
            <span
              className={`mr-2 inline-block rounded px-1.5 py-0.5 text-xs font-semibold uppercase ${
                a.severity === "critical"
                  ? "bg-red-100 text-red-800"
                  : "bg-amber-100 text-amber-800"
              }`}
            >
              {a.severity}
            </span>
            <span className="font-medium">
              {a.alert_type === "eval_regression"
                ? "Eval Regression"
                : "Critical Red Team Finding"}
            </span>
            <p className="mt-0.5 text-gray-600">{a.message}</p>
          </div>
          <button
            onClick={() => resolve(a.id)}
            className="ml-4 shrink-0 text-xs text-gray-500 underline hover:text-gray-800"
          >
            Resolve
          </button>
        </div>
      ))}
    </div>
  );
}
```
