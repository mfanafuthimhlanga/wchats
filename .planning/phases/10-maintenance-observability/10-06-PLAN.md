## Task 2 — test_observability_e2e.py

```python
"""Guarded E2E test for M10 Observability (OPS-05, OPS-06).

Requires: OPS_E2E_ENABLED=1, OPS_E2E_AGENT_ID=<uuid>, OPS_E2E_API_KEY=<key>.
"""
import os, pytest, httpx

OPS_E2E_ENABLED = os.environ.get("OPS_E2E_ENABLED", "0") == "1"
pytestmark = [
    pytest.mark.e2e,
    pytest.mark.skipif(not OPS_E2E_ENABLED, reason="OPS_E2E_ENABLED=1 required"),
]

BASE_URL = "http://localhost:8000/api/v1"
AGENT_ID = os.environ.get("OPS_E2E_AGENT_ID", "")
API_KEY = os.environ.get("OPS_E2E_API_KEY", "")
HEADERS = {"X-API-Key": API_KEY}


def test_ops01_alerts_endpoint_returns_list():
    """GET /agents/{id}/alerts returns 200 with a list."""
    r = httpx.get(f"{BASE_URL}/agents/{AGENT_ID}/alerts", headers=HEADERS)
    assert r.status_code == 200
    assert isinstance(r.json(), list)


def test_ops04_alert_resolve_roundtrip():
    """If an alert exists, resolve it and confirm it disappears from GET."""
    r = httpx.get(f"{BASE_URL}/agents/{AGENT_ID}/alerts", headers=HEADERS)
    alerts = r.json()
    if not alerts:
        pytest.skip("No active alerts — cannot test resolve roundtrip")
    alert_id = alerts[0]["id"]
    r2 = httpx.post(
        f"{BASE_URL}/agents/{AGENT_ID}/alerts/{alert_id}/resolve", headers=HEADERS
    )
    assert r2.status_code == 200
    r3 = httpx.get(f"{BASE_URL}/agents/{AGENT_ID}/alerts", headers=HEADERS)
    ids = [a["id"] for a in r3.json()]
    assert alert_id not in ids
```
